cmake_minimum_required(VERSION 3.24)
project(audit_log_service VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

include(FetchContent)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
set(HTTPLIB_INSTALL OFF CACHE BOOL "" FORCE)
set(HTTPLIB_USE_OPENSSL_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
set(HTTPLIB_USE_ZLIB_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
set(HTTPLIB_USE_BROTLI_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
set(HTTPLIB_USE_ZSTD_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
FetchContent_Declare(json
  URL https://codeload.github.com/nlohmann/json/tar.gz/65ee68451d8eb2b5f3a30b410476ab83deb3289b
  URL_HASH SHA256=13ef31d691947940a08909f8e0772f1d7d68e5da1678ee812a49c4bb0c996b2f)
FetchContent_Declare(httplib
  URL https://codeload.github.com/yhirose/cpp-httplib/tar.gz/4f3f9ef19be83ae97a5d9a059432dc00e445b7ab
  URL_HASH SHA256=415e8d638f6e2390f51565d4fdc86577dc0ef09f56da7ea74d7731d9da8349a6)
FetchContent_MakeAvailable(json httplib)
find_package(PostgreSQL REQUIRED)

file(GLOB_RECURSE CORE_SOURCES CONFIGURE_DEPENDS src/domain/*.cpp src/application/*.cpp)
add_library(audit_core ${CORE_SOURCES})
target_include_directories(audit_core PUBLIC include)
target_compile_options(audit_core PRIVATE -Wall -Wextra -Wpedantic)
file(GLOB_RECURSE PERSISTENCE_SOURCES CONFIGURE_DEPENDS src/persistence/*.cpp)
add_library(audit_persistence ${PERSISTENCE_SOURCES})
target_link_libraries(audit_persistence PUBLIC audit_core PostgreSQL::PostgreSQL nlohmann_json::nlohmann_json)
target_compile_options(audit_persistence PRIVATE -Wall -Wextra -Wpedantic)
file(GLOB_RECURSE API_SOURCES CONFIGURE_DEPENDS src/api/*.cpp)
add_library(audit_api ${API_SOURCES})
target_link_libraries(audit_api PUBLIC audit_core httplib::httplib nlohmann_json::nlohmann_json)
target_compile_options(audit_api PRIVATE -Wall -Wextra -Wpedantic)
add_executable(audit-log-service src/main.cpp)
target_link_libraries(audit-log-service PRIVATE audit_api audit_persistence)

include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(unit-tests tests/unit.cpp)
  target_link_libraries(unit-tests PRIVATE audit_core)
  add_test(NAME unit COMMAND unit-tests)
  add_test(NAME architecture COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/architecture.py)
endif()
